cmake_minimum_required(VERSION 3.21)
project(two_party_comparison LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(emp-tool 1.0 REQUIRED)
find_package(emp-ot 1.0 REQUIRED)
add_executable(modern-worker modern_worker.cpp)
target_compile_features(modern-worker PRIVATE cxx_std_20)
target_compile_definitions(modern-worker PRIVATE EMP_CONTEXT_CHECKS=1)
target_include_directories(modern-worker PRIVATE "${EMP_AG2PC_SOURCE}")
target_link_libraries(modern-worker PRIVATE emp-tool::emp-tool emp-ot::emp-ot)
add_executable(emp-nonlinear-state ../../handshake_backend/EmpHybridWork/nonlinear_state_fixture.cpp)
target_compile_features(emp-nonlinear-state PRIVATE cxx_std_20)
target_compile_definitions(emp-nonlinear-state PRIVATE EMP_CONTEXT_CHECKS=1)
target_include_directories(emp-nonlinear-state PRIVATE "${EMP_AG2PC_SOURCE}")
target_link_libraries(emp-nonlinear-state PRIVATE emp-tool::emp-tool emp-ot::emp-ot)
